\section{問題 1：無摂動基底状態}
%======================================================================

\begin{sol}
\textbf{(1)} $a=\tfrac1{\sqrt2}(u+\partial_u)$, $a^\dagger=\tfrac1{\sqrt2}(u-\partial_u)$ より
$[a,a^\dagger]=\tfrac12\bigl([u,-\partial_u]+[\partial_u,u]\bigr)=\tfrac12(1+1)=1$
（$[\partial_u,u]=1$）。また
\[
    a^\dagger a=\tfrac12(u-\partial_u)(u+\partial_u)=\tfrac12\bigl(u^2+u\partial_u-\partial_u u-\partial_u^2\bigr).
\]
$\partial_u u=1+u\partial_u$ ゆえ $u\partial_u-\partial_u u=-1$，よって
$a^\dagger a=\tfrac12(u^2-\partial_u^2-1)=H_0-\tfrac12$，すなわち $H_0=a^\dagger a+\tfrac12$。

\textbf{(2)} $a^\dagger a\ge0$ ゆえ最小固有値は $a\psi_0=0$ で実現される。
$(u+\partial_u)\psi_0=0\Rightarrow \psi_0'=-u\psi_0\Rightarrow \psi_0=Ce^{-u^2/2}$。
規格化 $\int C^2e^{-u^2}du=C^2\sqrt\pi=1$ から $C=\pi^{-1/4}$，すなわち
$\psi_0=\pi^{-1/4}e^{-u^2/2}$。エネルギーは $H_0\psi_0=(a^\dagger a+\tfrac12)\psi_0=\tfrac12\psi_0$，
ゆえ $\cE_0=\tfrac12$，次元を戻すと $E_0$ の最低次は $\cE_0\hb=\tfrac{\hb}{2}$（零点・1-loop energy），
$a_1=\tfrac12$。

\textbf{(3)} $\psi_0$ は $x=1$ の単一井戸に局在するガウシアンで，$x=-1$ の井戸やトンネルを含まない。
摂動級数はこの状態への非調和補正である。トンネル（インスタントン）は $e^{-S_0/\hb}$ で
$\hb$ の整級数に展開できず，どの有限次にも現れない。
\end{sol}

\section{問題 2：Bender--Wu 漸化式の導出}
%======================================================================

\begin{sol}
\textbf{(1)} $\partial_u(e^{-u^2/2}P)=e^{-u^2/2}(P'-uP)$ をもう一度微分すると
\[
    \partial_u^2(e^{-u^2/2}P)=e^{-u^2/2}\bigl(P''-2uP'+(u^2-1)P\bigr).
\]
ゆえ $\bigl(-\tfrac12\partial_u^2+\tfrac12u^2\bigr)(e^{-u^2/2}P)=e^{-u^2/2}\bigl(-\tfrac12P''+uP'+\tfrac12P\bigr)$。
非調和項 $\bigl(\tfrac12\epsilon u^3+\tfrac18\epsilon^2u^4\bigr)$ を加え，$\frac{H}{\hb}\psi=\cE\psi$ を
$e^{-u^2/2}$ で割ると
$-\tfrac12P''+uP'+\tfrac12P+\bigl(\tfrac12\epsilon u^3+\tfrac18\epsilon^2u^4\bigr)P=\cE P$，
すなわち \eqref{eq:Peq}。$P=1$ では左辺 $\tfrac12$，右辺 $\cE$ で $\cE=\tfrac12$ を再現
（無摂動基底状態に戻る）。

\textbf{(2)} $P=\sum_k\epsilon^kP_k$, $\cE-\tfrac12=\sum_{k\ge1}\cE_k\epsilon^k$ を \eqref{eq:Peq} に入れ $\epsilon^k$ を比較：
左辺 $-\tfrac12P''+uP'$ は $-\tfrac12P_k''+uP_k'$，$\tfrac12\epsilon u^3P$ は $\tfrac12u^3P_{k-1}$，
$\tfrac18\epsilon^2u^4P$ は $\tfrac18u^4P_{k-2}$，右辺は $\sum_{j=1}^k\cE_jP_{k-j}$。整理して \eqref{eq:BW}。
\end{sol}

\section{問題 3：低次係数の生成}
%======================================================================

\begin{sol}
左辺の演算子は $\bigl(-\tfrac12\partial_u^2+u\partial_u\bigr)u^m=m\,u^m-\tfrac12m(m-1)u^{m-2}$。
ゆえ $P_k=\sum_m c_m u^m$ に対し左辺の $u^m$ 係数は $m\,c_m-\tfrac12(m+2)(m+1)c_{m+2}$。
最高次から下げて解く。特に $u^0$ の式は $-c_2=[\text{右辺の }u^0]=\cE_k$，すなわち
\[
    \boxed{\;\cE_k=-[\,P_k\text{ の }u^2\text{ 係数}\,]\;}.
\]

\textbf{(1) $k=1$.}
右辺 $-\tfrac12u^3+\cE_1$。$P_1=au^3+bu$ の左辺は $3au^3+(b-3a)u$。比較：
$3a=-\tfrac12\Rightarrow a=-\tfrac16$；$b-3a=0\Rightarrow b=-\tfrac12$；$u^0:\cE_1=0$。
よって $P_1=-\tfrac16u^3-\tfrac12u$。

\textbf{(2) $k=2$.}
右辺 $=-\tfrac12u^3P_1-\tfrac18u^4+\cE_1P_1+\cE_2=\tfrac1{12}u^6+\tfrac18u^4+\cE_2$。
$P_2=cu^6+du^4+eu^2$ の左辺は $6cu^6+(4d-15c)u^4+(2e-6d)u^2-e$。比較：
$6c=\tfrac1{12}\Rightarrow c=\tfrac1{72}$；$4d-15c=\tfrac18\Rightarrow d=\tfrac1{12}$；
$2e-6d=0\Rightarrow e=\tfrac14$；$u^0:-e=\cE_2$。
ゆえ $P_2=\tfrac1{72}u^6+\tfrac1{12}u^4+\tfrac14u^2$，$\cE_2=-\tfrac14=a_2$。

\textbf{(3) $k=3,4$.}
$k=3$：$P_3$ は奇関数（$u^0$ 項なし）ゆえ $\cE_3=0$，
$P_3=-\tfrac1{1296}u^9-\tfrac1{144}u^7-\tfrac1{24}u^5-\tfrac18u^3-\tfrac14u$。
$k=4$：右辺 $=-\tfrac12u^3P_3-\tfrac18u^4P_2+\cE_2P_2+\cE_4$ の各冪は
$r_{12}=\tfrac1{2592},r_{10}=\tfrac1{576},r_8=\tfrac1{96},r_6=\tfrac1{36},r_4=\tfrac5{48},r_2=-\tfrac1{16}$。
$mc_m-\tfrac12(m+2)(m+1)c_{m+2}=r_m$ を上から解くと
$c_{12}=\tfrac1{31104},c_{10}=\tfrac1{2592},c_8=\tfrac1{288},c_6=\tfrac1{48},c_4=\tfrac5{48},c_2=\tfrac9{32}$，
最後に $-c_2=\cE_4$ から $\cE_4=-\tfrac9{32}=a_3$。同様に $a_4=\cE_6=-\tfrac{89}{128}$, $a_5=\cE_8=-\tfrac{5013}{2048}$。
\end{sol}

\section{問題 4：インスタントン作用}
%======================================================================

\begin{sol}
$|x|<1$ で $\sqrt{2V}=\sqrt{\tfrac14(1-x^2)^2}=\tfrac12(1-x^2)$ ゆえ
\[
    S_0=\int_{-1}^{1}\tfrac12(1-x^2)\,dx
    =\tfrac12\Bigl[x-\tfrac{x^3}{3}\Bigr]_{-1}^{1}=\tfrac12\Bigl(\tfrac23+\tfrac23\Bigr)=\frac23,
\]
よって $2S_0=\tfrac43$。（$\tfrac12\dot x^2=V$ より $S_0=\int\dot x^2d\tau=\int\sqrt{2V}\,dx$。）
\end{sol}

\section{問題 5：Gevrey-1 発散と収束半径}
%======================================================================

\begin{sol}
\textbf{(1) 構造的起源（漸化式から）。}
\eqref{eq:BW} で $\deg P_k=3k$。左辺を逆に解く各段は
$mc_m-\tfrac12(m+2)(m+1)c_{m+2}=r_m$，すなわち
$c_m=\tfrac1m\bigl[r_m+\tfrac12(m+2)(m+1)c_{m+2}\bigr]$ で，高次係数 $c_{m+2}$ が因子
$\tfrac{(m+2)(m+1)}{2m}\sim\tfrac{m}{2}$ を掛けながら $u^2$ 係数へ降りる。次数が $3k$ まで伸びるので
$\cE_k=-c_2$ はインデックスに比例する因子を多数蓄積し，階乗的に増大する。定量化は (2)。

\textbf{(2) 大次数解析（分散関係）。}
$E_0(\hb)$ の解析接続は I--$\bar{\mathrm I}$ により $\operatorname{Im}E_0\sim-Ce^{-2S_0/\hb}$（\eqref{eq:ImE}）を持つ。
分散関係 $a_n=\tfrac1\pi\int_0^\infty\frac{\operatorname{Im}E_0(\hb)}{\hb^{n+1}}d\hb$ に入れ，$t=1/\hb$ で
\[
    \int_0^\infty\frac{e^{-2S_0/\hb}}{\hb^{n+1}}d\hb=\int_0^\infty t^{n-1}e^{-2S_0t}\,dt=\frac{\Gamma(n)}{(2S_0)^n}
\]
ゆえ $a_n\sim-\tfrac{C}{\pi}\Gamma(n)(2S_0)^{-n}=-\tfrac{C}{\pi}(n-1)!\,(2S_0)^{-n}$，すなわち \eqref{eq:large}。
これは $(n-1)!$ 並みの増大――Gevrey-1 の階乗発散――の明示式で，スケールは I--$\bar{\mathrm I}$ 作用 $2S_0=4/3$。
（同形は漸近漸化式 $a_{n+1}/a_n\to n/(2S_0)$ の telescoping
$a_n=a_{n_0}\prod_{k=n_0}^{n-1}\tfrac{k}{2S_0}=K\Gamma(n)(2S_0)^{-n}$ でも得られる。）

\textbf{(3) 収束半径（系）。}
\eqref{eq:large} に直接ダランベールを適用すると
$\bigl|a_{n+1}/a_n\bigr|=\tfrac{\Gamma(n+1)}{\Gamma(n)}\tfrac1{2S_0}=\tfrac{n}{2S_0}\to\infty$，
ゆえ $R=\lim|a_n/a_{n+1}|=\lim\tfrac{2S_0}{n}=0$。収束半径は\emph{導いた}高次挙動の帰結であり，
低次の数項からの外挿ではない。

\textbf{(4) 整合性チェック（根拠ではない）。}
\eqref{eq:large} の予言は $|a_{n+1}/a_n|\simeq0.75n$。低次 5 項から
$|a_2/a_1|=0.5,\ |a_3/a_2|=1.125,\ |a_4/a_3|=\tfrac{89}{36}\approx2.472,\ |a_5/a_4|=\tfrac{5013}{1424}\approx3.520$
は予言値 $0.75,\,1.5,\,2.25,\,3.0$ に（$1/n$ 補正で上から）近づく。これは導出のチェックであって根拠ではない。
\end{sol}

\section{問題 6：Borel 変換}
%======================================================================

\begin{sol}
$a_n=K(n-1)!\,(2S_0)^{-n}$ では $b_n=a_n/(n-1)!=K(2S_0)^{-n}$ が公比 $1/(2S_0)$ の等比数列。ゆえ $|\zeta|<2S_0$ で
\[
    \Bor[E_0](\zeta)=\sum_{n\ge1}b_n\zeta^{n-1}
    =\frac{K}{2S_0}\sum_{n\ge1}\Bigl(\frac{\zeta}{2S_0}\Bigr)^{n-1}
    =\frac{K}{2S_0}\cdot\frac{1}{1-\zeta/2S_0}=\frac{K}{2S_0-\zeta}.
\]
最近接特異点は正実軸上の単純極 $\zeta=2S_0$。Borel 級数の収束半径は $2S_0$ で有限になり，
階乗発散が「除算」された。この極が Laplace 積分路 $(0,\infty)$ 上にあるため，横方向に避ける和の差は
留数 $\propto e^{-2S_0/\hb}$ の虚数的曖昧さ（=I--$\bar{\mathrm I}$ 非摂動効果）を与える。
\end{sol}

\section{問題 7：低次 Borel--Pad\'e}
%======================================================================

\begin{sol}
\textbf{(1)} $b_n=a_n/(n-1)!$ より
$b_1=\tfrac12,\ b_2=-\tfrac14,\ b_3=-\tfrac{9}{64},\ b_4=-\tfrac{89}{768},\ b_5=-\tfrac{5013}{49152}$。

\textbf{(2) $[1/1]$.}
$\frac{p_0+p_1\zeta}{1+q_1\zeta}$ を $b_1+b_2\zeta+b_3\zeta^2$ に合わせると $p_0=b_1$，$p_1-p_0q_1=b_2$，
$-p_1q_1+p_0q_1^2=b_3$。第 3 式に第 2 式を入れて $-q_1b_2=b_3$，$q_1=-b_3/b_2$。極は
$\zeta^\ast_{[1/1]}=-1/q_1=b_2/b_3=\frac{-1/4}{-9/64}=\frac{16}{9}\approx1.778$。

\textbf{(3) $[2/2]$.}
分母 $1+q_1\zeta+q_2\zeta^2$ は $b_4+q_1b_3+q_2b_2=0,\ b_5+q_1b_4+q_2b_3=0$ を満たす。
整理（第 1 式 $\times(-64)$，第 2 式 $\times(-768)$）すると
\[
    9q_1+16q_2=-\frac{89}{12},\qquad 89q_1+108q_2=-\frac{5013}{64}.
\]
解いて $q_2=\dfrac{8615}{86784}\approx0.09927$, $q_1=\tfrac19\bigl(-\tfrac{89}{12}-16q_2\bigr)\approx-1.00055$。
根は $\zeta=\dfrac{-q_1\pm\sqrt{q_1^2-4q_2}}{2q_2}=\dfrac{1.00055\pm0.77719}{0.19854}$，
すなわち $\zeta^\ast_{[2/2]}\approx1.125$（最近接）と $\approx8.95$。

\textbf{(4)} $[1/1]$ の $1.778$ と $[2/2]$ の $1.125$ は $2S_0=\tfrac43\approx1.333$ を上下から挟む。
Pad\'e の次数を上げると最近接極は $\zeta=2S_0$ へ収束する（$n\le33$ までの対角 Pad\'e $[15/15]$ で
$\zeta^\ast\approx1.3331\approx4/3$）。低次 5 項でも $\sim20\%$ で当たる。
\end{sol}

\section{問題 8：ループを閉じる}
%======================================================================

\begin{sol}
\textbf{(1)} 最近接特異点が $\zeta=\zeta^\ast$ なら $b_n\sim(\zeta^\ast)^{-n}$，逆 Borel
$a_n=(n-1)!\,b_n\sim\Gamma(n)(\zeta^\ast)^{-n}$。低次から当てた $\zeta^\ast\approx4/3=2S_0$ で，
これは問題 5 の大次数則 \eqref{eq:large}（増大率 $1/(2S_0)=3/4$）を再現する。
すなわち\emph{低次 5 項 $\to$ Pad\'e $\to$ 特異点 $\to$ 高次の増大率}。

\textbf{(2)} 独立計算のインスタントン作用 $S_0=2/3$（問題 4）の 2 倍 $2S_0=4/3$ が $\zeta^\ast$ と一致する。
無摂動基底状態 \eqref{eq:vac} への単純な補正計算に過ぎない摂動級数が，その低次の数項のうちに，
両井戸を結ぶトンネル（I--$\bar{\mathrm I}$）の非摂動スケール $2S_0$ を内包し，そこから高次の漸近挙動が
復元される。これが「摂動展開から非摂動効果が拾える」共鳴（resurgence）の核心であり，
摂動級数が自らの内に非摂動効果 $e^{-2S_0/\hb}$ を符号化していることの具体的な現れである。
\end{sol}
