\begin{abstract}
1 次元調和振動子に外場（source）$J(t)$ を結合させた系を経路積分で量子化する。
本演習の主眼は，\textbf{経路積分が「無限個の振動 mode それぞれについての 1 次元積分」に
ほかならない}ことを，具体的な計算を通じて体得することである。
揺らぎを演算子の固有関数で mode 展開すると作用が対角化され，
経路積分は mode 係数についての無限個の Fresnel（Gauss）積分の積になる。
この無限積を実行すると Feynman kernel が\textbf{結果として}現れる
――kernel を天下りに与えるのではなく，mode 積分から導くのが第 1 部の目標である。
第 2 部では，得られた kernel を真空状態で挟むと生成汎関数が得られ，
その指数部に Feynman 伝播関数が現れることを確かめる。
個々の積分はすべて Fresnel 積分と三角関数の恒等式に帰着し，
学部 2--3 年程度の物理数学で完結する。
解答は別ファイル \texttt{exercise\_generating\_functional\_solutions.tex} にある。
\end{abstract}

\section*{記号と準備}
\addcontentsline{toc}{section}{記号と準備}
%======================================================================
質量 $m$，角振動数 $\omega$ の 1 次元調和振動子に外場 $J(t)$ を結合させた Lagrangian
\begin{align}
    L
    &=
    \frac{1}{2}m\dot{q}^2
    -
    \frac{1}{2}m\omega^2q^2
    +
    qJ(t)
\label{eq:L}
\end{align}
を考える（$J=0$ で通常の調和振動子）。始時刻 $t_i$，終時刻 $t_f$，経過時間 $T:=t_f-t_i$ とし，
$0<\omega T<\pi$ を仮定する。本演習で用いる道具は次の 3 つだけである。

\begin{itemize}
\item \textbf{Fresnel 積分}（複素 $\alpha$ への Gauss 積分の拡張）：
\begin{align}
    \int_{-\infty}^{\infty}\! dx\; e^{-\alpha x^2}
    =\sqrt{\frac{\pi}{\alpha}}\quad(\Re\alpha\ge0),
    \qquad
    \int_{-\infty}^{\infty}\! dx\; e^{-\alpha x^2-2\beta x}
    =\sqrt{\frac{\pi}{\alpha}}\,e^{\beta^2/\alpha}\quad(\Re\alpha\ge0).
\label{eq:fresnel}
\end{align}
\item \textbf{Euler の無限乗積}：
\begin{align}
    \prod_{n=1}^{\infty}\left(1-\frac{z^2}{n^2}\right)=\frac{\sin\pi z}{\pi z}.
\label{eq:euler}
\end{align}
\item \textbf{自由粒子の Feynman kernel}（標準的な結果。$\omega\to0$ の規格化合わせにのみ用いる）：
\begin{align}
    K_{\text{free}}(x_f,t_f;x_i,t_i)
    =\left(\frac{m}{2\pi i\hbar T}\right)^{1/2}
    \exp\!\left(\frac{im(x_f-x_i)^2}{2\hbar T}\right).
\label{eq:free}
\end{align}
\end{itemize}

第 1 部（問題 \ref{prob:cl}--\ref{prob:kernel}）では Feynman kernel
\begin{align}
    K[J](x_f,t_f;x_i,t_i)
    :=
    \int_{q(t_i)=x_i}^{q(t_f)=x_f}\!\mathcal{D}q(t)\;
    \exp\!\left(\frac{i}{\hbar}S\right),
    \qquad
    S:=\int_{t_i}^{t_f}\! L\,dt
\label{eq:Kdef}
\end{align}
を mode 積分から構成する。第 2 部（問題 \ref{prob:W}--\ref{prob:Z}）では，
それを用いて生成汎関数を計算する。

\section{古典解と古典作用}
%======================================================================

\begin{prob}\label{prob:cl}
経路積分の鞍点となる古典解と，そこでの作用を求める。順に示せ。
\begin{enumerate}
\item \eqref{eq:L} の Euler--Lagrange 方程式が $m\ddot q+m\omega^2 q=J(t)$ となること。
\item 斉次方程式の一般解 $A\sin\omega(t-\alpha)$ に，特解
$\displaystyle\int_{t_i}^{t}\frac{\sin\omega(t-\tau)}{m\omega}J(\tau)\,d\tau$
を加えたものが一般解であること（特解が方程式を満たすことを，
$\dot q,\ddot q$ を計算して直接確認せよ）。
\item 境界条件 $q(t_i)=q_i,\ q(t_f)=q_f$ で定数 $A,\alpha$ を定めると，古典解は
\begin{align}
    q_{cl}(t)
    &=
    \frac{q_f\sin\omega(t-t_i)+q_i\sin\omega(t_f-t)}{\sin\omega T}
    -\frac{\sin\omega(t-t_i)}{\sin\omega T}
    \int_{t_i}^{t_f}\!\frac{\sin\omega(t_f-\tau)}{m\omega}J(\tau)\,d\tau
    \notag\\&\qquad
    +\int_{t_i}^{t}\!\frac{\sin\omega(t-\tau)}{m\omega}J(\tau)\,d\tau
\label{eq:qcl}
\end{align}
となること。
\item これを微分して，両端での速度が
\begin{subequations}\label{eq:qcldot}
\begin{align}
    \dot{q}_{cl}(t_i)
    &=
    \frac{\omega}{\sin\omega T}
    \left[\left(q_f-\int_{t_i}^{t_f}\!\frac{\sin\omega(t_f-\tau)}{m\omega}J(\tau)d\tau\right)
    -q_i\cos\omega T\right],
    \\
    \dot{q}_{cl}(t_f)
    &=
    \frac{\omega}{\sin\omega T}
    \left[\left(q_f-\int_{t_i}^{t_f}\!\frac{\sin\omega(t_f-\tau)}{m\omega}J(\tau)d\tau\right)\cos\omega T
    -q_i\right]
    +\int_{t_i}^{t_f}\!\frac{\cos\omega(t_f-\tau)}{m}J(\tau)d\tau
\end{align}
\end{subequations}
となること。
\item 作用 $S_{cl}[J]=\int_{t_i}^{t_f}L\,dt$ を部分積分し，運動方程式
$m\ddot q_{cl}+m\omega^2q_{cl}=J$ を使って境界項と source 項にまとめ，
\begin{align}
    S_{cl}[J]
    =\frac12 m\big(q_f\dot q_{cl}(t_f)-q_i\dot q_{cl}(t_i)\big)
    +\frac12\int_{t_i}^{t_f}\! q_{cl}(\tau)J(\tau)\,d\tau
\label{eq:Scl}
\end{align}
を得ること。
\end{enumerate}
\end{prob}

\section{揺らぎへの分離}
%======================================================================

\begin{prob}\label{prob:split}
古典解からのずれ $x(t):=q(t)-q_{cl}(t)$ を新しい積分変数にとる。
\begin{enumerate}
\item 任意の経路 $q$ は両端 $q(t_i)=q_i,\ q(t_f)=q_f$ で固定されているから，
$x$ は両端で $x(t_i)=x(t_f)=0$（Dirichlet 境界条件）を満たすことを確認せよ。
\item $q=q_{cl}+x$ を作用に代入して展開し，$x$ について 1 次の項が
\[
    \int_{t_i}^{t_f}\!\big(m\dot q_{cl}\dot x-m\omega^2 q_{cl}x+xJ\big)dt
\]
であることを示せ。さらに $\dot x$ を部分積分し（境界項は $x(t_i)=x(t_f)=0$ で消える），
古典運動方程式 $m\ddot q_{cl}+m\omega^2 q_{cl}-J=0$ を使うと，この 1 次の項が
\textbf{恒等的に消える}ことを示せ。
\item 結果として作用が
\begin{align}
    S=S_{cl}[J]+S_{\text{fl}}[x],
    \qquad
    S_{\text{fl}}[x]:=\frac{m}{2}\int_{t_i}^{t_f}\!\big(\dot x^2-\omega^2 x^2\big)dt
\label{eq:Ssplit}
\end{align}
と\textbf{完全に分離}することを示せ。揺らぎ $S_{\text{fl}}$ には source $J$ が現れず，
$J$ 依存性はすべて $S_{cl}[J]$ に押し込められている点に注意せよ。
\item したがって kernel \eqref{eq:Kdef} は
\begin{align}
    K[J]
    =\exp\!\left(\frac{i}{\hbar}S_{cl}[J]\right)
    \int_{x(t_i)=0}^{x(t_f)=0}\!\mathcal{D}x\;
    \exp\!\left(\frac{i}{\hbar}S_{\text{fl}}[x]\right)
\label{eq:Kfactor}
\end{align}
と，古典寄与（指数の肩）と揺らぎの経路積分の積に書けることを示せ。
\textbf{残った経路積分は $J$ にも $q_i,q_f$ にもよらない}ことを確認せよ
（以降この量を $\mathcal{N}(T)$ と書く）。
\end{enumerate}
\end{prob}

%======================================================================
\section{経路積分を mode 積分に翻訳する}
%======================================================================
ここが本演習の核心である。揺らぎの経路積分 $\mathcal{N}(T)$ を，
「すべての経路 $x(t)$ にわたる積分」から「すべての mode 係数にわたる積分」へ翻訳する。

\begin{prob}\label{prob:mode}
\begin{enumerate}
\item $S_{\text{fl}}$ を部分積分して（$x$ は両端で消える）
\begin{align}
    S_{\text{fl}}[x]
    =-\int_{t_i}^{t_f}\! x(t)\,\hat O\,x(t)\,dt,
    \qquad
    \hat O:=\frac{m}{2}\left(\frac{d^2}{dt^2}+\omega^2\right)
\label{eq:OP}
\end{align}
の形に書けることを示せ。
\item 境界条件 $x(t_i)=x(t_f)=0$ のもとでの固有値問題 $\hat O\phi_n=\lambda_n\phi_n$ を解き，
規格直交完全系
\begin{align}
    \phi_n(t)=\sqrt{\frac{2}{T}}\sin\omega_n(t-t_i),
    \quad
    \omega_n=\frac{n\pi}{T},
    \quad
    \lambda_n=\frac{m}{2}\big(\omega^2-\omega_n^2\big)
    \quad(n=1,2,\dots)
\label{eq:modes}
\end{align}
が得られることを示せ（$\int_{t_i}^{t_f}\phi_n\phi_m\,dt=\delta_{nm}$ も確認）。
\item 任意の揺らぎを $x(t)=\sum_{n=1}^{\infty}a_n\phi_n(t)$ と展開すると，
固有関数の直交性から作用が\textbf{対角化}され
\begin{align}
    S_{\text{fl}}[x]=-\sum_{n=1}^{\infty}\lambda_n a_n^2
\label{eq:Sdiag}
\end{align}
となることを示せ。各 mode は互いに独立な変数 $a_n$ になった。
\item 経路積分の測度は，経路 $x(t)$ から mode 係数 $\{a_n\}$ への線形変換であるから，
（$a_n$ によらない）Jacobian $\mathcal{J}$ を用いて
$\mathcal{D}x=\mathcal{J}\prod_{n=1}^{\infty}da_n$ と書ける。これを使い，
\textbf{揺らぎの経路積分が mode 係数についての無限個の 1 次元 Fresnel 積分の積}
\begin{align}
    \mathcal{N}(T)
    =\int_{x(t_i)=0}^{x(t_f)=0}\!\mathcal{D}x\;e^{iS_{\text{fl}}/\hbar}
    =\mathcal{J}\prod_{n=1}^{\infty}\left[\int_{-\infty}^{\infty}\! da_n\;
    \exp\!\left(-\frac{i}{\hbar}\lambda_n a_n^2\right)\right]
\label{eq:Nmode}
\end{align}
になることを示せ。これが「経路積分は無限回の mode 積分である」という主張の数式表現である。
\end{enumerate}
\end{prob}

\section{無限個の Fresnel 積分から Feynman kernel へ}
%======================================================================

\begin{prob}\label{prob:kernel}
\eqref{eq:Nmode} の無限積を実行し，Feynman kernel を構成する。順に示せ。
\begin{enumerate}
\item 各 mode の Fresnel 積分 \eqref{eq:fresnel}（$\Re(i\lambda_n)\ge0$ は $i\epsilon$ 処方で保証される）が
\[
    \int_{-\infty}^{\infty}\! da_n\;\exp\!\left(-\frac{i}{\hbar}\lambda_n a_n^2\right)
    =\sqrt{\frac{\pi\hbar}{i\lambda_n}}
    =\sqrt{\frac{2\pi i\hbar}{m\omega_n^2}\,\frac{1}{1-\omega^2/\omega_n^2}}
\]
を与えることを示せ。
\item 無限積をとり，$\omega$ 依存部分と非依存部分を分離して
\begin{align}
    \mathcal{N}(T)
    =\mathcal{J}\sqrt{\prod_{n=1}^{\infty}\frac{2\pi i\hbar}{m\omega_n^2}}\;
    \times\;\sqrt{\prod_{n=1}^{\infty}\frac{1}{1-\omega^2/\omega_n^2}}
\end{align}
と書けることを示し，Euler の乗積 \eqref{eq:euler}（$z=\omega T/\pi$）を用いて
$\displaystyle\prod_{n=1}^{\infty}\Big(1-\frac{\omega^2}{\omega_n^2}\Big)=\frac{\sin\omega T}{\omega T}$
を示せ。
\item $\displaystyle\mathcal{J}\sqrt{\prod_n\frac{2\pi i\hbar}{m\omega_n^2}}$ は $\omega$ にも $J$ にもよらない。
そこで $\omega\to0$（自由粒子）の極限で値を固定する。$\omega\to0$ では
$\sqrt{\omega T/\sin\omega T}\to1$ かつ $S_{cl}\to$ 自由粒子の古典作用となるから，
\eqref{eq:Kfactor} を自由粒子の kernel \eqref{eq:free} と比較して
\begin{align}
    \mathcal{J}\sqrt{\prod_{n=1}^{\infty}\frac{2\pi i\hbar}{m\omega_n^2}}
    =\left(\frac{m}{2\pi i\hbar T}\right)^{1/2}
\end{align}
を示せ。
\item 以上を合わせて，揺らぎの経路積分が
$\displaystyle\mathcal{N}(T)=\Big(\frac{m\omega}{2\pi i\hbar\sin\omega T}\Big)^{1/2}$
となり，\textbf{無限個の mode 積分の結果として}Feynman kernel
\begin{align}
    K[J](x_f,t_f;x_i,t_i)
    =\left(\frac{m\omega}{2\pi i\hbar\sin\omega T}\right)^{1/2}
    \exp\!\left(\frac{i}{\hbar}S_{cl}[J]\right)
\label{eq:kernel}
\end{align}
が現れることを示せ。
\end{enumerate}
\end{prob}

\section{生成汎関数}
%======================================================================
第 2 部では，第 1 部で構成した kernel \eqref{eq:kernel} を用いる。
基底状態（真空）の波動関数は調和振動子の基底状態（energy $\tfrac12\hbar\omega$）
\begin{align}
    \phi_0(t,q)
    =\left(\frac{m\omega}{\pi\hbar}\right)^{1/4}
    \exp\!\left(-\frac{m\omega}{2\hbar}q^2-\frac{i\omega t}{2}\right)
\label{eq:vac}
\end{align}
である。始・終状態をともに真空にとった生成汎関数を
\begin{align}
    W_{00}[J](T)
    :=\int dq_f\,dq_i\;\phi_0^*(t_i+T,q_f)\,\phi_0(t_i,q_i)\,K[J](q_f,t_i+T;q_i,t_i)
\label{eq:Wdef}
\end{align}
で定義する（$J=0$ なら真空$\to$真空の遷移振幅）。これを最後まで計算する。

\begin{prob}\label{prob:W}
\begin{enumerate}
\item \eqref{eq:vac} と \eqref{eq:kernel} を \eqref{eq:Wdef} に代入し，
波動関数の位相が $e^{i\omega T/2}$ にまとまることを確かめたうえで，被積分関数の指数部を
$q_i,q_f$ について整理して
\begin{align}
    W_{00}[J](T)
    =\frac{m\omega}{\pi\hbar(2i\sin\omega T)^{1/2}}\,
    e^{i\omega T/2+E}
    \int dq_f\,dq_i\;
    e^{-A(q_i^2+q_f^2)-2Bq_iq_f-2Cq_f-2Dq_i}
\label{eq:WABCD}
\end{align}
の形にせよ。各係数を $J$ の汎関数として求め，次になることを示せ：
\begin{align*}
    A&=\frac{m\omega}{2\hbar}\frac{-ie^{i\omega T}}{\sin\omega T},
    &
    B&=\frac{m\omega}{2\hbar}\frac{i}{\sin\omega T},
    &
    F&:=\int_{t_i}^{t_i+T}\! e^{i\omega\tau}J(\tau)\,d\tau,
    \\
    C&=\frac{1}{2i\hbar}\frac{\Im[e^{-i\omega t_i}F]}{\sin\omega T},
    &
    D&=\frac{1}{2i\hbar}\frac{\Im[e^{i\omega(t_i+T)}F^*]}{\sin\omega T},
\end{align*}
$E$ は $J$ について 2 次の（$q_i,q_f$ 非依存の）項全体。
\item $q_f$，続いて $q_i$ について Fresnel 積分 \eqref{eq:fresnel} を 2 回実行し
（収束条件 $\Re A\ge0,\ \Re(A^2-B^2)\ge0$ を確認），
\begin{align}
    W_{00}[J](T)
    =\frac{m\omega}{\hbar[2i\sin\omega T(A^2-B^2)]^{1/2}}\,
    \exp\!\left(\frac{i\omega T}{2}+E+\frac{A(C^2+D^2)-2BCD}{A^2-B^2}\right)
\label{eq:Wgauss}
\end{align}
を導け。
\item $A^2-B^2=\big(\frac{m\omega}{\hbar}\big)^2\frac{-ie^{i\omega T}}{2\sin\omega T}$ を示し，
\eqref{eq:Wgauss} の前因子が $e^{-i\omega T/2}$ に簡約されることを示せ。
\item 補助量 $E,\,C^2+D^2,\,BCD$ を $F,F^*,|F|^2$ で表し，これらを合わせると
$F^2,{F^*}^2,|F|^2$ の境界項が\textbf{相殺}して
\begin{align}
    E+\frac{A(C^2+D^2)-2BCD}{A^2-B^2}
    =-\frac{1}{2\hbar m\omega}
    \int_{t_i}^{t_i+T}\!\!d\tau\int_{t_i}^{t_i+T}\!\!d\tau'\,
    J(\tau)J(\tau')\,\theta(\tau-\tau')\,e^{i\omega(\tau'-\tau)}
\label{eq:exponent}
\end{align}
となること（$\theta$ は階段関数）を示せ。これが最も計算量の多い小問である。
\end{enumerate}
\end{prob}

\section{Feynman 伝播関数}
%======================================================================

\begin{prob}\label{prob:DF}
\begin{enumerate}
\item 前因子 $e^{-i\omega T/2}$ と位相 $e^{i\omega T/2}$ が相殺することを使い，
さらに \eqref{eq:exponent} の被積分関数を $\tau\leftrightarrow\tau'$ について対称化して，
生成汎関数が
\begin{align}
    W_{00}[J](T)
    =\exp\!\left(\frac{i}{4\hbar}
    \int_{t_i}^{t_i+T}\!\!d\tau\int_{t_i}^{t_i+T}\!\!d\tau'\,
    J(\tau)J(\tau')\,D_F(\tau-\tau')\right),
\label{eq:Wfinal}
\end{align}
\begin{align}
    D_F(\tau-\tau')
    :=\frac{i}{m\omega}
    \big[\theta(\tau-\tau')\,e^{-i\omega(\tau-\tau')}+\theta(\tau'-\tau)\,e^{i\omega(\tau-\tau')}\big]
\label{eq:DF}
\end{align}
と書けることを示せ。
\item 階段関数の Fourier 表示
$\theta(s)=-\int\frac{d\omega'}{2\pi i}\frac{e^{-i\omega's}}{\omega'+i\epsilon}$ を用いて，
$D_F$ が
\begin{align}
    D_F(\tau-\tau')
    =\frac{2}{m}\int\frac{d\omega'}{2\pi}
    \frac{e^{-i\omega'(\tau-\tau')}}{-\omega'^2+\omega^2-i\epsilon}
\label{eq:DFfourier}
\end{align}
と書けることを示せ。
\item これより $D_F$ が作用の演算子 $\hat O=\frac{m}{2}(\frac{d^2}{dt^2}+\omega^2)$ の Green 関数
\begin{align}
    \frac{m}{2}\left(\frac{d^2}{dt^2}+\omega^2\right)D_F(t)=\delta(t)
\end{align}
であることを示せ。第 1 部で揺らぎを支配していたまさにその演算子の逆が，
第 2 部で伝播関数として現れたことになる。
\end{enumerate}
\end{prob}

\section{経路積分による統一的理解}
%======================================================================

\begin{prob}\label{prob:unified}
第 1 部と第 2 部を貫く論理を，経路積分の言葉でまとめる。
\begin{enumerate}
\item kernel の経路積分表示を \eqref{eq:Wdef} に戻すと，$q_i,q_f$ 積分と波動関数が
端点の重ね合わせ $\bra{\phi_0},\ket{\phi_0}$ を与え，生成汎関数が端点を固定しない全経路積分
\[
    W_{00}[J](T)=\bra{\phi_0}\int_{\forall q(t)}\!\mathcal{D}q\;e^{iS/\hbar}\ket{\phi_0}
\]
になることを示せ。
\item source 付き調和振動子の作用が，部分積分により
\[
    \frac{i}{\hbar}S
    =-\frac12\int d\tau\,q(\tau)\,\frac{i}{\hbar}m\Big(\frac{d^2}{d\tau^2}+\omega^2\Big)q(\tau)
    +\int d\tau\,q(\tau)\,\frac{i}{\hbar}J(\tau)
\]
と書けることを確かめよ。これは「$-\frac12 x\!\cdot\!Ax+J\!\cdot\!x$」型の\textbf{無限次元 Gauss 積分}
そのものである。
\item 有限次元の source 付き Gauss 積分の公式
\[
    \int\prod_i\frac{dx_i}{\sqrt{2\pi}}\,e^{-\frac12 x_iA_{ij}x_j+x_iJ_i}
    =\frac{1}{\sqrt{\det A}}\exp\!\Big(\tfrac12 J_i(A^{-1})_{ij}J_j\Big)
\]
を無限次元へ拡張し，行列 $A\to\frac{i}{\hbar}m(\partial_\tau^2+\omega^2)$，
逆行列 $A^{-1}\to$ Green 関数 $D_F$ と読み替えると，\eqref{eq:Wfinal} が
\[
    W[J]=\exp\!\left[\frac{i}{4\hbar}\int d\tau\,d\tau'\,J(\tau')
    \Big\{\tfrac{m}{2}\big(\partial_\tau^2+\omega^2\big)\Big\}^{-1}J(\tau)\right]
\]
として自然に再現されることを説明せよ。
mode 展開（第 1 部）で $A$ を対角化したものが固有値 $\lambda_n$，
その逆 $A^{-1}$ を座標表示したものが $D_F$ である，という対応を述べよ。
\end{enumerate}
\end{prob}

\section{発展：分配関数と統計力学}

\begin{prob}\label{prob:Z}
生成汎関数 $W_{00}[J]$ の計算から波動関数 $\phi_0$ の寄与を除き，
周期境界条件 $x:=q_i=q_f$ を課すと分配関数
$Z[J](\tau)=\int dx\,K_E(\tau;x,x)$ が得られる（$K_E$ は虚時間 $\tau=iT$ の kernel）。
\begin{enumerate}
\item 問題 \ref{prob:W} とほぼ同じ計算で，$q_i=q_f=x$ の 1 変数 Fresnel 積分が
\begin{align}
    Z[J](\tau)
    =\frac{1}{e^{i\omega T/2}-e^{-i\omega T/2}}\,\exp\!\left(\frac{H^2}{2G}\right),
    \quad
    G=A+B-\frac{m\omega}{2\hbar},\quad H=C+D
\end{align}
を与えることを示せ（波動関数の寄与とは $A$ に含まれる $\frac{m\omega}{2\hbar}$）。
\item $J=0$ では $H=0$ ゆえ，虚時間 $\tau=iT=\hbar\beta$ とおくと
\[
    Z[J=0]=\frac{1}{e^{\beta\hbar\omega/2}-e^{-\beta\hbar\omega/2}}
    =\frac{1}{2\sinh(\beta\hbar\omega/2)}
    =\sum_{n=0}^{\infty}e^{-\beta\hbar\omega(n+1/2)}
\]
となり，統計力学で習う調和振動子の分配関数を再現することを確かめよ。
\end{enumerate}
\end{prob}
