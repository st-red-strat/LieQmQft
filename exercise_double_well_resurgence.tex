\begin{abstract}
1 次元二重井戸量子力学を題材に，古典論の多重インスタントン解と，
量子摂動論の発散構造（Borel 特異性）との対応を，紙と鉛筆だけで辿る。
\textbf{前半（古典）}では，真空解とそれらを結ぶインスタントン解，さらにそれらを重ね合わせた
多重インスタントン配位の性質と作用を調べる。\textbf{後半（量子）}では，片方の井戸の無摂動基底状態から
出発し，そこからの補正を生成する Bender--Wu 漸化式を導いて低次係数を求め，その大次数挙動と
摂動級数の収束性，Borel 変換の特異点構造を解析する。最後に，低次の数項だけから Borel 特異点を
どこまで復元できるかを Borel--Pad\'e で確かめ，\emph{摂動展開から非摂動効果が拾える}という
共鳴（resurgence）の核心に至る。\textbf{各問は表式・数値・オーダーを答えさせる形にしてあり，
本文中には原則として解答を与えていない}（正しく計算できていれば表式が一致するはずである）。
完全な解答は別ファイル \texttt{exercise\_double\_well\_resurgence\_solutions.tex} にある。
\end{abstract}

\section{設定}
%======================================================================
質量 $1$，Planck 定数 $\hb$ を陽に残した 1 次元 Hamiltonian
\begin{align}
    H=\frac{p^2}{2}+V(x),\qquad V(x)=\frac18\bigl(x^2-1\bigr)^2
\label{eq:H}
\end{align}
を考える。$V$ は $x=\pm1$ に縮退した極小（$V=0$）を持つ対称二重井戸である。
ユークリッド時間 $\tau$（$t=-i\tau$）での作用は
$S_E=\int d\tau\,\bigl[\tfrac12\dot x^2+V(x)\bigr]$（$\dot{}=d/d\tau$）で，
運動方程式は $\ddot x=V'(x)=\tfrac12 x(x^2-1)$ である。
片方の極小 $x=1$ のまわりで $x=1+y$ とおくと $x^2-1=2y+y^2$ より
\begin{align}
    V=\frac18(2y+y^2)^2=\underbrace{\tfrac12 y^2}_{\text{調和}（\omega=1）}
    +\underbrace{\tfrac12 y^3+\tfrac18 y^4}_{\text{非調和}}.
\label{eq:Vexp}
\end{align}
基底エネルギーを $\hb$ で展開した摂動級数を $E_0(\hb)=\sum_{n\ge1}a_n\hb^n$ と書く。
後半（問題 \ref{prob:vac} 以降）でこの係数 $a_n$ を無摂動基底状態からの補正として生成し，
その大次数挙動を解析する。前半（問題 \ref{prob:classical}）ではまず古典解を調べる。

\section{古典解：真空・インスタントン・多重インスタントン}
%======================================================================

\begin{prob}\label{prob:classical}
ユークリッド運動方程式 $\ddot x=V'(x)=\tfrac12 x(x^2-1)$ の有限作用配位を調べる。
\begin{enumerate}
\item \textbf{真空解。}運動方程式を満たす定数解 $x(\tau)\equiv x_v$ をすべて求めよ。そのうち作用 $S_E$ が
有限となるのはどれか，またそのときの $S_E$ の値を求めよ。
\item \textbf{インスタントン解。}有限作用解はエネルギー保存 $\tfrac12\dot x^2-V=0$（finite action ゆえ
積分定数 $=0$）を満たす。$\dot x=\sqrt{2V}=\tfrac12(1-x^2)$ を積分して，$-1$ から $+1$ へ移る解
$x_{\mathrm{cl}}(\tau)$（中心位置 $\tau_0$ を含む）を$\tau$ の初等関数として求めよ。
さらに $\tau\to+\infty$ での $1-x_{\mathrm{cl}}(\tau)$ の主要な振る舞いを $\tau$ の関数として求め，
その大きさのオーダーをランダウ記号で表せ。$x_{\mathrm{cl}}$ の符号を変えたものは
$+1$ から $-1$ へ移る反インスタントンである。
\item \textbf{作用。}インスタントン 1 個の作用
$S_0=\int_{-\infty}^{\infty}\dot x_{\mathrm{cl}}^2\,d\tau=\int_{-1}^{1}\sqrt{2V}\,dx
=\int_{-1}^{1}\tfrac12(1-x^2)\,dx$
を実行し，$S_0$ の値を求めよ。
\item \textbf{作用密度とそのグラフ。}オンシェル $\tfrac12\dot x^2=V$ ゆえ Euclid Lagrangian 密度は
$\varepsilon(\tau):=\tfrac12\dot x_{\mathrm{cl}}^2+V=\dot x_{\mathrm{cl}}^2$。
(2) で求めた $x_{\mathrm{cl}}$ から $\varepsilon(\tau)$ を $\tau$ の初等関数として求めよ。
$\tau_0=0$ として $x_{\mathrm{cl}}(\tau)$ と $\varepsilon(\tau)$ のグラフを同じ $\tau$ 軸上に描き，
\textbf{$\varepsilon$ のピーク値}，\textbf{$|\tau|\to\infty$ での $\varepsilon$ の減衰のオーダー（ランダウ記号）}，
\textbf{$\displaystyle\int_{-\infty}^\infty\varepsilon\,d\tau$ の値}を求めよ。
このグラフから読み取れる $\varepsilon$ の $\tau$ 軸上での広がり方が，次問の作用の加法性の根拠になる。
\item \textbf{多重インスタントン。}中心 $\tau_1\ll\tau_2\ll\cdots\ll\tau_N$ に，インスタントンと反インスタントンを
交互につないだ配位を考える。前問の $\varepsilon$ の振る舞いから，よく離れた配位の作用密度は山の和
$\varepsilon(\tau)\approx\sum_i\varepsilon(\tau-\tau_i)$ で近似でき，隣接する山どうしの重なり（相互作用）の
大きさを隣接間隔 $R=\tau_{i+1}-\tau_i$ の関数（オーダーをランダウ記号）で表せ。
分離 $R\to\infty$ の極限でこの配位は運動方程式を（核の外で厳密に，全体として極限・弱の意味で）満たし，
両端で 2 つの真空に漸近する弱解となる。\textbf{この $N$ 個の連なりの作用を（相互作用が消える極限で）
$N$ と $S_0$ で表せ}。とくに，同じ真空へ戻る配位（基底状態の振幅に効くもの）は(反)インスタントンが
偶数個 $2N$ 個並んだもので，\textbf{その作用を $N,S_0$ で表せ}。
\end{enumerate}
\end{prob}

\section{場の理論的解釈：tadpole 和と別の真空}
%======================================================================
インスタントンが\emph{なぜ非摂動効果なのか}を，場の理論の言葉で見る。
$x=1$ の真空のまわり $y=x-1$ では \eqref{eq:Vexp} の $V(y)=\tfrac12y^2+\tfrac12y^3+\tfrac18y^4$ を用い，
非調和項 $\tfrac12y^3,\tfrac18y^4$ を頂点とする摂動論をすべて $\langle y\rangle=0$ のまわりで展開する。
別の真空 $x=-1$ は $y=-2$ にある。

\begin{prob}\label{prob:tadpole}
\begin{enumerate}
\item \textbf{古典臨界点。}$V'(y)=y+\tfrac32y^2+\tfrac12y^3$ を因数分解し，$V'(y)=0$ の解をすべて求めよ。
各解で $V''(y)$ の符号を調べ，極小か極大かを判定せよ。
\item \textbf{source による期待値（tadpole 和）。}作用に source 項 $+Jy$ を加えると，
（古典・tree レベルの）期待値 $\langle y\rangle=y_*(J)$ は $V'(y_*)=J$ で定まる。これは図的には，
伝播子 $1/V''(0)=1/\omega^2$ に source $J$ をつなぎ，頂点 $\tfrac12y^3,\tfrac18y^4$ で補正した
tree 図（tadpole）を\emph{無限個足し上げた}ものである。$y_*(J)$ を $y=0$ の真空のまわりで
$y_*(J)=\sum_{k\ge1}\gamma_k J^k$ と展開し，\textbf{係数 $\gamma_1,\gamma_2,\gamma_3$ を求めよ}。
\item \textbf{この級数はどこまで届くか。}この級数は $J\to0$ で $y_*\to0$ となる枝（$y=0$ の真空に連続的に
つながる枝）だけを表す。その収束半径は $V''(y_c)=0$ となる最近接の分岐点で決まる。
\textbf{$y_c$，収束半径 $|J_c|=|V'(y_c)|$，およびこの枝で到達できる $|y|$ の最大値を求めよ}。
得た最大値を，障壁の頂上 $y=-1$ と別の真空 $y=-2$ の位置と比較し，tadpole 級数が
それらに到達できるか否かを論ぜよ。
\item \textbf{結論。}別の真空へ移るには障壁を越えねばならず，それを行うのが前節のユークリッド・
インスタントンである。その重みは結合定数のベキではなく $e^{-S_0/\hb}$ で，上の摂動（tadpole）級数の
どの有限次にも現れない。すなわち真空間遷移（インスタントン）が非摂動効果であることを，
(2)(3) を踏まえて論ぜよ。にもかかわらず後半で見るように，摂動級数の大次数挙動はこの非摂動スケールを
内包する――これが共鳴（resurgence）の要点である。
\end{enumerate}
\end{prob}

\noindent（前半（古典・場の理論的考察）はここまで。以降，量子論の摂動級数 $a_n$ を生成し，
その発散構造に上の作用が現れることを見る。）

\section{無摂動基底状態（調和近似の波動関数）}
%======================================================================
$y=\sqrt{\hb}\,u$ と再スケールすると Hamiltonian は
\begin{align}
    \frac{H}{\hb}=\underbrace{\Bigl(-\tfrac12\partial_u^2+\tfrac12u^2\Bigr)}_{H_0}
    +\epsilon\,\tfrac12u^3+\epsilon^2\,\tfrac18u^4,\qquad \epsilon:=\sqrt{\hb}.
\label{eq:Hresc}
\end{align}
固有値を $\cE:=E_0/\hb=\sum_{k\ge0}\cE_k\epsilon^k$ と展開すると，$E_0=\hb\,\cE=\sum_k\cE_k\hb^{1+k/2}$ より
\begin{align}
    a_n=\cE_{2(n-1)}\qquad(\text{奇数次 }\cE_{2k+1}\text{ はパリティから消える}).
\label{eq:map}
\end{align}
摂動はすべて無摂動部 $H_0$（$x=1$ の井戸の調和振動子）の基底状態を出発点とする。

\begin{prob}\label{prob:vac}
$H_0=-\tfrac12\partial_u^2+\tfrac12u^2$ の基底状態を求める。
\begin{enumerate}
\item 演算子 $a:=\tfrac1{\sqrt2}(u+\partial_u)$, $a^\dagger:=\tfrac1{\sqrt2}(u-\partial_u)$ について
交換子 $[a,a^\dagger]$ を計算し，$H_0$ を $a^\dagger a$ で表せ（$\partial_u u=1+u\partial_u$）。
\item 基底状態は $a\psi_0=0$ で定まる。これを解いて規格化した $\psi_0(u)$ を求め，
無摂動基底エネルギー $\cE_0$ の値（したがって $a_1$ の値）を求めよ。
\item \textbf{物理的意味.}この $\psi_0$ は $x=1$ の片方の井戸に局在し，もう一方の井戸 $x=-1$ や
両井戸間のトンネル（インスタントン）を含まない。以降の摂動級数はこの単一井戸状態への補正であり，
トンネル効果 $\propto e^{-S_0/\hb}$ はどの有限次にも現れない――にもかかわらず，級数の\emph{発散の仕方}が
その非摂動スケールを内包することを以下で見る。
\end{enumerate}
\end{prob}

\section{Bender--Wu 漸化式：基底状態への補正}
%======================================================================

\begin{prob}\label{prob:bw}
補正を取り込んだ基底状態を，無摂動部に補正因子を掛けた形
$\psi(u)=e^{-u^2/2}P(u)$, $P=\sum_{k\ge0}\epsilon^kP_k$, $P_0=1$ で書く
（$\epsilon^0$ 部分 $e^{-u^2/2}$ が無摂動基底状態，$P_{k\ge1},\cE_{k\ge1}$ が補正）。
\begin{enumerate}
\item $\partial_u^2(e^{-u^2/2}P)=e^{-u^2/2}(P''-2uP'+(u^2-1)P)$ を用いて，
固有値方程式 $\frac{H}{\hb}\psi=\cE\psi$ を $P$ についての方程式に書き直し，
\begin{align}
    -\tfrac12 P''+uP'+\Bigl(\tfrac12\epsilon u^3+\tfrac18\epsilon^2 u^4\Bigr)P=\Bigl(\cE-\tfrac12\Bigr)P
\label{eq:Peq}
\end{align}
を導け。$P=1$ のとき $\cE$ がいくつになるかを確かめよ。
\item \eqref{eq:Peq} を $\epsilon$ で展開し（$\cE-\tfrac12=\sum_{k\ge1}\cE_k\epsilon^k$），$\epsilon^k$ の係数を比較して
$P_k$ の漸化式
\begin{align}
    -\tfrac12 P_k''+uP_k'=-\tfrac12 u^3 P_{k-1}-\tfrac18 u^4 P_{k-2}+\sum_{j=1}^{k}\cE_j\,P_{k-j}
\label{eq:BW}
\end{align}
を導け。各 $k$ で右辺は既知の $P_{<k}$ から定まり，左辺の演算子 $-\tfrac12\partial_u^2+u\partial_u$ を
逆に解いて $P_k$ が，可解条件（定数項の整合）から $\cE_k$ が定まる。
\end{enumerate}
\end{prob}

\section{漸化式で低次係数を求める}
%======================================================================

\begin{prob}\label{prob:gen}
正規化 $P_k(0)=0$（$k\ge1$）のもとで \eqref{eq:BW} を低次から解く。
左辺は単項式に対し $\bigl(-\tfrac12\partial_u^2+u\partial_u\bigr)u^m=m\,u^m-\tfrac12 m(m-1)u^{m-2}$ と作用する。
\begin{enumerate}
\item \eqref{eq:BW} の $u^0$ の係数を比較して，$\cE_k$ を $P_k$ の特定の係数で表せ。
\item $k=1$：$P_1=au^3+bu$ と置いて \eqref{eq:BW} を解き，$P_1(u)$ と $\cE_1$ を求めよ。
\item $k=2$：$P_2=cu^6+du^4+eu^2$ と置いて $P_2(u)$ を求め，$\cE_2$（$=a_2$）の値を求めよ。
\item さらに漸化式を続けて（$P_3,P_4$ を経て）$\cE_3$ と $\cE_4$（$=a_3$）の値を求めよ。
\end{enumerate}
（より高次は機械的に続けられるが手計算は煩雑なので，後の Borel--Pad\'e で必要な
$a_4=\cE_6,\ a_5=\cE_8$ は問題 \ref{prob:pade} で数値を与える。）
\end{prob}

\section{大次数挙動と収束半径}
%======================================================================
低次の数項の比から「発散しそうだ」と外挿するのは，高次の主張を低次から演繹する不当な推論である。
ここでは大次数挙動\emph{そのもの}を導き，収束半径をその\emph{帰結}として決める。

\begin{prob}\label{prob:asym}
\begin{enumerate}
\item \textbf{増大の構造的起源（漸化式から）。}\eqref{eq:BW} で $\deg P_k=3k$ は $k$ に比例して増え，
左辺を逆に解く各段 $mc_m-\tfrac12(m+2)(m+1)c_{m+2}=r_m$ では，高次係数 $c_{m+2}$ に因子
$\tfrac{(m+2)(m+1)}{2m}$ が掛かりながら $u^2$ 係数へ降りる。この構造から，$\cE_k=-[u^2\text{ 係数}]$ が
$k$ とともにどのようなオーダーで増大するかを定性的に見積もれ（定量化は (2)）。
\item \textbf{増大率と閉じた形（大次数解析：分散関係＋インスタントン）。}$E_0(\hb)$ を解析接続すると，
両井戸を結ぶ I--$\bar{\mathrm I}$ インスタントンにより指数的に小さい虚部
$\operatorname{Im}E_0(\hb)\sim-Ce^{-2S_0/\hb}$（$\hb\to0^+$；$C>0$ は定数，$2S_0$ は問題 \ref{prob:classical}(5) の
I--$\bar{\mathrm I}$ 作用）が生じる。摂動係数は分散関係
$a_n=\tfrac1\pi\int_0^\infty\frac{\operatorname{Im}E_0(\hb)}{\hb^{n+1}}d\hb$ で与えられる。
これに $\operatorname{Im}E_0$ を代入し，$t=1/\hb$ と置換して\textbf{積分を実行し，$a_n$ の大次数の主要形を
$n,\,S_0,\,C$ で表せ}。さらにその増大のオーダーをランダウ記号（$\Gamma$ または階乗を用いて）で表し，
増大率を支配するスケールを答えよ。
\item \textbf{系：収束半径。}(2) で導いた漸近形にダランベールの判定法
$R=\lim_{n\to\infty}|a_n/a_{n+1}|$ を適用し，\textbf{$|a_{n+1}/a_n|$ の $n\to\infty$ での振る舞いと
収束半径 $R$ を求めよ}。収束半径が，低次の数項の外挿ではなく導いた高次挙動の帰結であることに注意せよ。
\item \textbf{整合性チェック（根拠ではない）。}問題 \ref{prob:gen} で求めた $a_1,a_2,a_3$ から
比 $|a_2/a_1|,|a_3/a_2|$ を計算し，(2) が予言する漸近線 $|a_{n+1}/a_n|\simeq n/(2S_0)$ と比較せよ
（低次では $1/n$ 補正があるので一致は粗い）。これは導出の整合性チェックであって収束半径の根拠ではない。
\end{enumerate}
\end{prob}

\section{Borel 変換と特異点}
%======================================================================

\begin{prob}\label{prob:borel}
摂動級数の Borel 変換を
$\displaystyle\Bor[E_0](\zeta):=\sum_{n\ge1}\frac{a_n}{(n-1)!}\,\zeta^{\,n-1}$
で定める（形式的に $\int_0^\infty e^{-\zeta/\hb}\zeta^{n-1}d\zeta=(n-1)!\,\hb^n$ で元に戻る）。
\begin{enumerate}
\item 問題 \ref{prob:asym}(2) の大次数形を用いて Borel 係数 $b_n:=a_n/(n-1)!$ の大次数での振る舞いを
$n,\,S_0$ で表せ。
\item これを使って $\Bor[E_0](\zeta)=\sum_{n\ge1}b_n\zeta^{n-1}$ を閉じた形に和し，
\textbf{得られた関数の最近接特異点の位置（$S_0$ で表す）と型（極か分岐か）を求めよ}。
\item その特異点が Laplace 積分路 $\zeta\in(0,\infty)$ 上にあることが，Borel 和の曖昧さ（非摂動効果）の
起源である。この曖昧さの大きさのオーダーを $\hb$ の関数（$e$ の肩を $S_0,\hb$ で）として表せ。
\end{enumerate}
\end{prob}

\section{多重インスタントンと Borel 特異点列}
%======================================================================
最近接特異点は 1 組の I--$\bar{\mathrm I}$ 対（作用 $2S_0$）に対応していた。
古典前半で見た\emph{多重}インスタントン（$2N$ 個，作用 $2NS_0$）は，量子論で何を生むか。

\begin{prob}\label{prob:periodic}
\begin{enumerate}
\item Borel 和 $E_0=\int_0^\infty e^{-\zeta/\hb}\Bor[E_0](\zeta)d\zeta$ において，Borel 平面の特異点 $\zeta=A$ は
非摂動因子 $e^{-A/\hb}$ に対応する（$\int^\infty$ の鞍点）。問題 \ref{prob:classical}(5) の同じ真空へ戻る
$2N$-インスタントン弱解（作用 $2NS_0$）が非摂動振幅 $\propto e^{-2NS_0/\hb}$ を与えることから，
\textbf{各 $N$ が $\Bor[E_0]$ に与える特異点の位置 $\zeta_N$ を $N,S_0$ で表せ}。これら $\{\zeta_N\}$ の
$\zeta$ 軸上の並び方を述べ，\textbf{隣接間隔を $S_0$ で表せ}。さらに $S_0$ の値を入れて最初の 3 点の数値を求めよ。
\item \textbf{大次数への寄与。}$N$ 番目の特異点 $\zeta_N=2NS_0$ は，大次数係数 $a_n$ に
$\propto\Gamma(n)\,\zeta_N^{-n}$ の項を与える（$b_n=a_n/(n-1)!$ には $\propto\zeta_N^{-n}$）。
最近接 $N=1$ の項に対する $N=2$ の項の比 $\dfrac{\zeta_2^{-n}}{\zeta_1^{-n}}$ を $n$ と $S_0$ の関数として簡約せよ。
これから $n\to\infty$ でどの $N$ の項が大次数を支配するかを述べよ。
\item \textbf{対応の要約。}古典の $2N$-インスタントン弱解（作用 $2NS_0$）と，量子 Borel 平面の特異点列
$\{\zeta_N\}$ がどう対応するかを述べよ。
\end{enumerate}
\end{prob}

\section{低次 Borel--Pad\'e：準備と精度の予想}
%======================================================================
Borel 特異点を，\textbf{低次の数項だけ}からどこまで復元できるかを，Borel 級数の Pad\'e 近似
（有理関数近似）の極で見る。

\paragraph{なぜ Pad\'e（有理関数）が要るのか。}
Borel 級数を有限項 $N$ で打ち切ると $\Bor_N(\zeta)=\sum_{n=1}^N b_n\zeta^{n-1}$ は\emph{多項式}（整関数）で，
特異点を一切持たない。これを項別に Laplace 積分すれば
$\int_0^\infty e^{-\zeta/\hb}\Bor_N(\zeta)d\zeta=\sum_{n=1}^N b_n(n-1)!\,\hb^n=\sum_{n=1}^N a_n\hb^n$
と，元の（発散）級数の打ち切りに\emph{厳密に}戻るだけで，新しい情報（特異点・非摂動的曖昧さ）は何も出ない。
特異点は\textbf{無限項の和ではじめて現れる}。有限の係数からそれを捉えるには，多項式の外へ
\emph{解析接続}せねばならない――分母の零点として実際の極を作る有理関数近似（Pad\'e）はそのための最小の道具である。

\paragraph{Pad\'e 近似の定義と係数の決め方。}
ベキ級数 $f(\zeta)=\sum_{j\ge0}c_j\zeta^j$ の \textbf{$[L/M]$ Pad\'e 近似}とは，分子・分母の次数が
$\deg P\le L$, $\deg Q\le M$，$Q(0)=1$ の有理関数 $\dfrac{P(\zeta)}{Q(\zeta)}$ で，その Taylor 展開が
$f$ と $\zeta^{L+M}$ 次まで一致するもの：
\begin{align}
    \frac{P(\zeta)}{Q(\zeta)}=f(\zeta)+O(\zeta^{L+M+1}),\qquad
    P=\sum_{i=0}^{L}p_i\zeta^i,\quad Q=1+\sum_{i=1}^{M}q_i\zeta^i .
\end{align}
両辺に $Q$ を掛けた $P=Qf+O(\zeta^{L+M+1})$ の係数比較から，まず\textbf{分母}は
$\zeta^{L+1},\dots,\zeta^{L+M}$ 次が消える条件
\begin{align}
    \sum_{i=0}^{M}q_i\,c_{L+k-i}=0\quad(k=1,\dots,M;\ q_0=1)
\label{eq:padeden}
\end{align}
という $M$ 元連立 1 次方程式で $q_1,\dots,q_M$ が定まり，つづいて\textbf{分子}は
$p_i=\sum_{j=0}^{i}q_j\,c_{i-j}$（$i=0,\dots,L$）で定まる。
近似の\textbf{極}は分母の零点 $Q(\zeta)=0$ である。
本問では $f=\Bor[E_0]=\sum_{n\ge1}b_n\zeta^{n-1}$ なので $c_j=b_{j+1}$（$j\ge0$）と読み替える。

まず，最近接特異点 $2S_0$ の\emph{次}の特異点が果たす役割を見て，低次 Pad\'e の精度を理論的に\emph{予想}しておく。

\begin{prob}\label{prob:rate}
\begin{enumerate}
\item \textbf{次の特異点の寄与（定量）。}問題 \ref{prob:periodic}(2) より，最近接特異点 $2S_0$ と
次の特異点 $4S_0$ が大次数係数 $b_n=a_n/(n-1)!$ に与える項は，それぞれ $\propto(2S_0)^{-n}$，$\propto(4S_0)^{-n}$ である。
後者の前者に対する比 $\dfrac{(4S_0)^{-n}}{(2S_0)^{-n}}$ を $n$ の関数として簡約し，$n=2,3,4,5$ で数値評価せよ。
これが速く減衰するか否かを述べよ。
\item \textbf{Pad\'e 精度の理論的予想。}有限の係数から最近接特異点 $\zeta_1=2S_0$ を当てる Pad\'e の
極位置の相対誤差は，次の特異点 $\zeta_2=4S_0$ までの距離で制限され，Pad\'e の次数を 1 つ上げるごとに概ね
\emph{距離比 $\zeta_1/\zeta_2$}倍に改善すると期待される（これは Pad\'e 収束の一般論で，ここでは認めてよい）。
この予想される 1 次数あたりの改善因子 $\zeta_1/\zeta_2=2S_0/4S_0$ を数値で求めよ。
（この理論的予想を，次節で実際に Pad\'e を計算して検証する。）
\end{enumerate}
\end{prob}

\section{Pad\'e による特異点の同定と精度の検証}
%======================================================================

\begin{prob}\label{prob:pade}
\begin{enumerate}
\item 問題 \ref{prob:gen} で求めた $a_1,a_2,a_3$ から Borel 係数 $b_n=a_n/(n-1)!$（$n=1,2,3$）を求めよ。
\item \textbf{$[1/1]$ Pad\'e} $\dfrac{p_0+p_1\zeta}{1+q_1\zeta}$ を $b_1,b_2,b_3$ に合わせ，その極 $\zeta^\ast_{[1/1]}$ を
まず $b_n$ で表し，次に数値を求めよ。これを問題 \ref{prob:borel} の最近接特異点 $2S_0$ と比べよ。
\item \textbf{$[2/2]$ Pad\'e}：これには $a_4=\cE_6=-\dfrac{89}{128}$, $a_5=\cE_8=-\dfrac{5013}{2048}$
（漸化式を続けると得られる；ここでは入力として与える）も使う。$b_4,b_5$ を求め，
分母 $1+q_1\zeta+q_2\zeta^2$ を $b_1,\dots,b_5$ に合わせる連立 1 次方程式
$b_4+q_1b_3+q_2b_2=0,\ b_5+q_1b_4+q_2b_3=0$ を解いて，最近接の極 $\zeta^\ast_{[2/2]}$ の数値を求めよ。
\item 求めた $\zeta^\ast_{[1/1]},\zeta^\ast_{[2/2]}$ と $2S_0$ との位置関係を述べよ。Pad\'e の次数を上げると
最近接極がどこへ向かうと期待されるかを述べよ。
\item \textbf{相対誤差。}$\zeta^\ast_{[1/1]},\zeta^\ast_{[2/2]}$ の，最近接特異点 $2S_0$ に対する相対誤差
$|\zeta^\ast-2S_0|/(2S_0)$ をそれぞれ求め，$[2/2]$ と $[1/1]$ の誤差の比を求めよ。
\item \textbf{予想との比較。}(5) で得た誤差比を，問題 \ref{prob:rate}(2) で\emph{予想}した 1 次数あたりの
改善因子と比較し，オーダーが一致するかを論ぜよ。これが，低次（$n\le5$）の Pad\'e でも $2S_0$ がかなり
当たる理由であることを説明せよ（参考：$[15/15]$ では相対誤差 $\sim2\times10^{-4}$）。
\end{enumerate}
\end{prob}

\section{ループを閉じる：摂動 $\Rightarrow$ 非摂動}
%======================================================================

\begin{prob}\label{prob:loop}
\begin{enumerate}
\item 問題 \ref{prob:pade} で\emph{低次の数項だけ}から得た Borel 特異点 $\zeta^\ast$ から，逆 Borel
$a_n=(n-1)!\,b_n\sim\Gamma(n)(\zeta^\ast)^{-n}$ により高次の増大率が決まる。これが問題 \ref{prob:asym} で
導いた大次数則と一致することを確かめよ。
\item この $\zeta^\ast$ が，前半で独立に求めたインスタントン作用 $S_0$ とどう関係するか，さらに特異点列
$\{\zeta_N\}$（問題 \ref{prob:periodic}）が古典多重インスタントン弱解とどう対応するかを総括せよ。
無摂動基底状態への単純な補正計算に過ぎない摂動級数が，その低次の数項のうちに，両井戸を結ぶトンネル
（多重インスタントン）の非摂動スケールを内包する――これが「\textbf{摂動展開から非摂動効果が拾える}」という
共鳴（resurgence）の核心である。問題 \ref{prob:tadpole} で見たとおりインスタントン自体は摂動では到達できない
非摂動配位であり，それでもなおその作用が摂動級数の発散に刻まれている点が決定的である，と論ぜよ。
\end{enumerate}
\end{prob}
